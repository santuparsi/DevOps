# Matrix Strategy Pipeline
# Tests application across multiple OS, Node versions, and browsers

trigger:
  branches:
    include:
      - main
      - develop

# ==========================================
# STAGE 1: Multi-Platform Build & Test
# ==========================================
stages:
  - stage: BuildAndTest
    displayName: 'Build and Test Across Platforms'
    jobs:
      # Job 1: Matrix build across different OS and Node versions
      - job: MultiPlatformBuild
        displayName: 'Build on'
        strategy:
          matrix:
            Linux_Node16:
              imageName: 'ubuntu-latest'
              nodeVersion: '16.x'
              platform: 'linux'
            Linux_Node18:
              imageName: 'ubuntu-latest'
              nodeVersion: '18.x'
              platform: 'linux'
            Linux_Node20:
              imageName: 'ubuntu-latest'
              nodeVersion: '20.x'
              platform: 'linux'
            Windows_Node18:
              imageName: 'windows-latest'
              nodeVersion: '18.x'
              platform: 'windows'
            MacOS_Node18:
              imageName: 'macOS-latest'
              nodeVersion: '18.x'
              platform: 'macos'
        pool:
          vmImage: $(imageName)
        
        steps:
          - task: NodeTool@0
            displayName: 'Install Node.js $(nodeVersion)'
            inputs:
              versionSpec: $(nodeVersion)
          
          - script: |
              node --version
              npm --version
            displayName: 'Display Node and NPM versions'
          
          - script: npm install
            displayName: 'npm install'
          
          - script: npm run build
            displayName: 'Build application'
          
          - script: npm test
            displayName: 'Run unit tests'
            continueOnError: true
          
          - task: PublishTestResults@2
            displayName: 'Publish test results'
            condition: succeededOrFailed()
            inputs:
              testResultsFormat: 'JUnit'
              testResultsFiles: '**/test-results.xml'
              testRunTitle: 'Tests - $(platform) - Node $(nodeVersion)'

      # Job 2: Browser compatibility testing
      - job: BrowserTests
        displayName: 'Browser Compatibility Tests'
        dependsOn: MultiPlatformBuild
        condition: succeeded()
        pool:
          vmImage: 'ubuntu-latest'
        strategy:
          matrix:
            Chrome:
              browserName: 'chrome'
              browserVersion: 'latest'
            Firefox:
              browserName: 'firefox'
              browserVersion: 'latest'
            Edge:
              browserName: 'edge'
              browserVersion: 'latest'
            Safari:
              browserName: 'safari'
              browserVersion: 'latest'
        
        steps:
          - task: NodeTool@0
            inputs:
              versionSpec: '18.x'
          
          - script: npm install
            displayName: 'Install dependencies'
          
          - script: |
              echo "Running E2E tests on $(browserName)"
              npm run test:e2e -- --browser=$(browserName)
            displayName: 'Run E2E tests on $(browserName)'
          
          - task: PublishTestResults@2
            condition: succeededOrFailed()
            inputs:
              testResultsFormat: 'JUnit'
              testResultsFiles: '**/e2e-results-$(browserName).xml'
              testRunTitle: 'E2E Tests - $(browserName)'

  # ==========================================
  # STAGE 2: Database Compatibility Testing
  # ==========================================
  - stage: DatabaseTests
    displayName: 'Database Compatibility Tests'
    dependsOn: BuildAndTest
    condition: succeeded()
    jobs:
      - job: DatabaseMatrix
        displayName: 'Test with'
        strategy:
          matrix:
            PostgreSQL_13:
              databaseType: 'postgres'
              databaseVersion: '13'
              connectionString: 'postgres://user:pass@localhost:5432/testdb'
            PostgreSQL_15:
              databaseType: 'postgres'
              databaseVersion: '15'
              connectionString: 'postgres://user:pass@localhost:5432/testdb'
            MySQL_8:
              databaseType: 'mysql'
              databaseVersion: '8.0'
              connectionString: 'mysql://user:pass@localhost:3306/testdb'
            MongoDB_6:
              databaseType: 'mongo'
              databaseVersion: '6.0'
              connectionString: 'mongodb://localhost:27017/testdb'
        pool:
          vmImage: 'ubuntu-latest'
        
        services:
          database:
            image: $(databaseType):$(databaseVersion)
            ports:
              - 5432:5432  # Postgres
              - 3306:3306  # MySQL
              - 27017:27017  # MongoDB
        
        steps:
          - task: NodeTool@0
            inputs:
              versionSpec: '18.x'
          
          - script: npm install
            displayName: 'Install dependencies'
          
          - script: |
              echo "Testing with $(databaseType) version $(databaseVersion)"
              export DATABASE_URL=$(connectionString)
              npm run test:integration
            displayName: 'Run integration tests'
          
          - task: PublishTestResults@2
            condition: succeededOrFailed()
            inputs:
              testResultsFormat: 'JUnit'
              testResultsFiles: '**/integration-results.xml'
              testRunTitle: 'Integration - $(databaseType) $(databaseVersion)'

  # ==========================================
  # STAGE 3: Security and Performance Testing
  # ==========================================
  - stage: SecurityAndPerformance
    displayName: 'Security & Performance'
    dependsOn: DatabaseTests
    condition: succeeded()
    jobs:
      - job: SecurityScan
        displayName: 'Security Scanning'
        pool:
          vmImage: 'ubuntu-latest'
        steps:
          - script: npm install
            displayName: 'Install dependencies'
          
          - script: npm audit --production
            displayName: 'NPM Security Audit'
            continueOnError: true
          
          - script: |
              npx snyk test --severity-threshold=high
            displayName: 'Snyk Vulnerability Scan'
            continueOnError: true
            env:
              SNYK_TOKEN: $(SNYK_TOKEN)
          
          - task: SonarCloudPrepare@1
            displayName: 'Prepare SonarCloud'
            inputs:
              SonarCloud: 'SonarCloud-Connection'
              organization: 'myorg'
              scannerMode: 'CLI'
          
          - task: SonarCloudAnalyze@1
            displayName: 'Run SonarCloud Analysis'
          
          - task: SonarCloudPublish@1
            displayName: 'Publish Quality Gate Result'

      - job: PerformanceTest
        displayName: 'Performance Testing'
        pool:
          vmImage: 'ubuntu-latest'
        steps:
          - script: npm install
            displayName: 'Install dependencies'
          
          - script: npm run build
            displayName: 'Build application'
          
          - script: |
              echo "Running load tests..."
              npx artillery run load-test.yml
            displayName: 'Artillery Load Testing'
          
          - script: |
              echo "Running Lighthouse performance audit..."
              npm install -g @lhci/cli
              lhci autorun
            displayName: 'Lighthouse Performance Audit'
          
          - task: PublishBuildArtifacts@1
            displayName: 'Publish Performance Reports'
            inputs:
              PathtoPublish: 'performance-reports'
              ArtifactName: 'performance-results'

  # ==========================================
  # STAGE 4: Conditional Production Deploy
  # ==========================================
  - stage: Production
    displayName: 'Deploy to Production'
    dependsOn: SecurityAndPerformance
    condition: |
      and(
        succeeded(),
        eq(variables['Build.SourceBranch'], 'refs/heads/main'),
        eq(variables['Build.Reason'], 'Manual')
      )
    jobs:
      - deployment: DeployProduction
        displayName: 'Production Deployment'
        environment: 'production'
        pool:
          vmImage: 'ubuntu-latest'
        strategy:
          runOnce:
            deploy:
              steps:
                - script: echo "Deploying to production..."
                  displayName: 'Deploy Application'
