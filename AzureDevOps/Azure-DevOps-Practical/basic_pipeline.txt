# Basic CI Pipeline for Node.js Application
# Triggers on every push to main branch

trigger:
  branches:
    include:
      - main
      - develop
  paths:
    exclude:
      - README.md
      - docs/*

# Use Ubuntu agent from Microsoft-hosted pool
pool:
  vmImage: 'ubuntu-latest'

# Pipeline variables
variables:
  nodeVersion: '18.x'
  buildConfiguration: 'Release'

# Pipeline stages
stages:
  - stage: Build
    displayName: 'Build and Test'
    jobs:
      - job: BuildJob
        displayName: 'Build Application'
        steps:
          # Step 1: Setup Node.js
          - task: NodeTool@0
            displayName: 'Install Node.js'
            inputs:
              versionSpec: $(nodeVersion)
          
          # Step 2: Install dependencies
          - script: |
              npm install
            displayName: 'npm install'
          
          # Step 3: Run linting
          - script: |
              npm run lint
            displayName: 'Run ESLint'
            continueOnError: true
          
          # Step 4: Run unit tests
          - script: |
              npm run test -- --coverage
            displayName: 'Run Unit Tests'
          
          # Step 5: Publish test results
          - task: PublishTestResults@2
            displayName: 'Publish Test Results'
            condition: succeededOrFailed()
            inputs:
              testResultsFormat: 'JUnit'
              testResultsFiles: '**/test-results.xml'
              failTaskOnFailedTests: true
          
          # Step 6: Publish code coverage
          - task: PublishCodeCoverageResults@2
            displayName: 'Publish Code Coverage'
            inputs:
              codeCoverageTool: 'Cobertura'
              summaryFileLocation: '$(System.DefaultWorkingDirectory)/coverage/cobertura-coverage.xml'
          
          # Step 7: Build application
          - script: |
              npm run build
            displayName: 'Build Application'
          
          # Step 8: Archive build output
          - task: ArchiveFiles@2
            displayName: 'Archive Build Files'
            inputs:
              rootFolderOrFile: '$(System.DefaultWorkingDirectory)/dist'
              includeRootFolder: false
              archiveType: 'zip'
              archiveFile: '$(Build.ArtifactStagingDirectory)/$(Build.BuildId).zip'
          
          # Step 9: Publish artifact for deployment
          - task: PublishBuildArtifacts@1
            displayName: 'Publish Build Artifact'
            inputs:
              PathtoPublish: '$(Build.ArtifactStagingDirectory)'
              ArtifactName: 'drop'
              publishLocation: 'Container'
